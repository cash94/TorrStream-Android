# Proguard rules specific to the FFmpeg extension.
# (TorrStream: stock rules of Media3 1.11.1 plus the video decoder.)

# This prevents the names of native methods from being obfuscated. The classes are
# found by name in JNI_OnLoad and their natives bound with RegisterNatives.

    native <methods>;
}
